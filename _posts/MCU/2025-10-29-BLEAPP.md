## Step 2. NU40 DK, 납땜

보드에 ADC 핀과, GND 핀에 연결해줄 전선과 납땜 해주어야 합니다.

static void uart_cb(const struct device *dev, struct uart_event *evt, void *user_data)
{
	ARG_UNUSED(dev);

	static size_t aborted_len;
	struct uart_data_t *buf;
	static uint8_t *aborted_buf;
	static bool disable_req;

	switch (evt->type) {
	case UART_TX_DONE:
		LOG_DBG("UART_TX_DONE");
		if ((evt->data.tx.len == 0) ||
		    (!evt->data.tx.buf)) {
			return;
		}

		if (aborted_buf) {
			buf = CONTAINER_OF(aborted_buf, struct uart_data_t,
					   data[0]);
			aborted_buf = NULL;
			aborted_len = 0;
		} else {
			buf = CONTAINER_OF(evt->data.tx.buf, struct uart_data_t,
					   data[0]);
		}

		k_free(buf);

		buf = k_fifo_get(&fifo_uart_tx_data, K_NO_WAIT);
		if (!buf) {
			return;
		}

		if (uart_tx(uart, buf->data, buf->len, SYS_FOREVER_MS)) {
			LOG_WRN("Failed to send data over UART");
		}

		break;

	case UART_RX_RDY:
		LOG_DBG("UART_RX_RDY");
		buf = CONTAINER_OF(evt->data.rx.buf, struct uart_data_t, data[0]);
		buf->len += evt->data.rx.len;

		if (disable_req) {
			return;
		}

		if ((evt->data.rx.buf[buf->len - 1] == '\n') ||
		    (evt->data.rx.buf[buf->len - 1] == '\r')) {
			disable_req = true;
			uart_rx_disable(uart);
		}

		break;

	case UART_RX_DISABLED:
		LOG_DBG("UART_RX_DISABLED");
		disable_req = false;

		buf = k_malloc(sizeof(*buf));
		if (buf) {
			buf->len = 0;
		} else {
			LOG_WRN("Not able to allocate UART receive buffer");
			k_work_reschedule(&uart_work, UART_WAIT_FOR_BUF_DELAY);
			return;
		}

		uart_rx_enable(uart, buf->data, sizeof(buf->data),
			       UART_WAIT_FOR_RX);

		break;

	case UART_RX_BUF_REQUEST:
		LOG_DBG("UART_RX_BUF_REQUEST");
		buf = k_malloc(sizeof(*buf));
		if (buf) {
			buf->len = 0;
			uart_rx_buf_rsp(uart, buf->data, sizeof(buf->data));
		} else {
			LOG_WRN("Not able to allocate UART receive buffer");
		}

		break;

	case UART_RX_BUF_RELEASED:
		LOG_DBG("UART_RX_BUF_RELEASED");
		buf = CONTAINER_OF(evt->data.rx_buf.buf, struct uart_data_t,
				   data[0]);

		if (buf->len > 0) {
			k_fifo_put(&fifo_uart_rx_data, buf);
		} else {
			k_free(buf);
		}

		break;

	case UART_TX_ABORTED:
		LOG_DBG("UART_TX_ABORTED");
		if (!aborted_buf) {
			aborted_buf = (uint8_t *)evt->data.tx.buf;
		}

		aborted_len += evt->data.tx.len;
		buf = CONTAINER_OF((void *)aborted_buf, struct uart_data_t,
				   data);

		uart_tx(uart, &buf->data[aborted_len],
			buf->len - aborted_len, SYS_FOREVER_MS);

		break;

	default:
		break;
	}
}

static void uart_work_handler(struct k_work *item)
{
	struct uart_data_t *buf;

	buf = k_malloc(sizeof(*buf));
	if (buf) {
		buf->len = 0;
	} else {
		LOG_WRN("Not able to allocate UART receive buffer");
		k_work_reschedule(&uart_work, UART_WAIT_FOR_BUF_DELAY);
		return;
	}

	uart_rx_enable(uart, buf->data, sizeof(buf->data), UART_WAIT_FOR_RX);
}

static bool uart_test_async_api(const struct device *dev)
{
	const struct uart_driver_api *api =
			(const struct uart_driver_api *)dev->api;

	return (api->callback_set != NULL);
}

static int uart_init(void)
{
	int err;
	int pos;
	struct uart_data_t *rx;
	struct uart_data_t *tx;

	if (!device_is_ready(uart)) {
		return -ENODEV;
	}

	if (IS_ENABLED(CONFIG_USB_DEVICE_STACK)) {
		err = usb_enable(NULL);
		if (err && (err != -EALREADY)) {
			LOG_ERR("Failed to enable USB");
			return err;
		}
	}

	rx = k_malloc(sizeof(*rx));
	if (rx) {
		rx->len = 0;
	} else {
		return -ENOMEM;
	}

	k_work_init_delayable(&uart_work, uart_work_handler);


	if (IS_ENABLED(CONFIG_UART_ASYNC_ADAPTER) && !uart_test_async_api(uart)) {
		/* Implement API adapter */
		uart_async_adapter_init(async_adapter, uart);
		uart = async_adapter;
	}

	err = uart_callback_set(uart, uart_cb, NULL);
	if (err) {
		k_free(rx);
		LOG_ERR("Cannot initialize UART callback");
		return err;
	}

	if (IS_ENABLED(CONFIG_UART_LINE_CTRL)) {
		LOG_INF("Wait for DTR");
		while (true) {
			uint32_t dtr = 0;

			uart_line_ctrl_get(uart, UART_LINE_CTRL_DTR, &dtr);
			if (dtr) {
				break;
			}
			/* Give CPU resources to low priority threads. */
			k_sleep(K_MSEC(100));
		}
		LOG_INF("DTR set");
		err = uart_line_ctrl_set(uart, UART_LINE_CTRL_DCD, 1);
		if (err) {
			LOG_WRN("Failed to set DCD, ret code %d", err);
		}
		err = uart_line_ctrl_set(uart, UART_LINE_CTRL_DSR, 1);
		if (err) {
			LOG_WRN("Failed to set DSR, ret code %d", err);
		}
	}

	tx = k_malloc(sizeof(*tx));

	if (tx) {
		pos = snprintf(tx->data, sizeof(tx->data),
			       "Starting Nordic UART service example\r\n");

		if ((pos < 0) || (pos >= sizeof(tx->data))) {
			k_free(rx);
			k_free(tx);
			LOG_ERR("snprintf returned %d", pos);
			return -ENOMEM;
		}

		tx->len = pos;
	} else {
		k_free(rx);
		return -ENOMEM;
	}

	err = uart_tx(uart, tx->data, tx->len, SYS_FOREVER_MS);
	if (err) {
		k_free(rx);
		k_free(tx);
		LOG_ERR("Cannot display welcome message (err: %d)", err);
		return err;
	}

	err = uart_rx_enable(uart, rx->data, sizeof(rx->data), UART_WAIT_FOR_RX);
	if (err) {
		LOG_ERR("Cannot enable uart reception (err: %d)", err);
		/* Free the rx buffer only because the tx buffer will be handled in the callback */
		k_free(rx);
	}

	return err;
}


static void connected(struct bt_conn *conn, uint8_t err)
{
	char addr[BT_ADDR_LE_STR_LEN];

	if (err) {
		LOG_ERR("Connection failed, err 0x%02x %s", err, bt_hci_err_to_str(err));
		return;
	}

	bt_addr_le_to_str(bt_conn_get_dst(conn), addr, sizeof(addr));
	LOG_INF("Connected %s", addr);

	current_conn = bt_conn_ref(conn);
	connection_ready = false; // MTU 교환 대기
	dk_set_led_on(CON_STATUS_LED);
	LOG_INF("Waiting for MTU exchange and subscription...");
	
	// 연결 전에 큐에 쌓인 데이터 모두 비우기
	struct uart_data_t *buf;
	while ((buf = k_fifo_get(&fifo_uart_rx_data, K_NO_WAIT)) != NULL) {
		LOG_INF("Dropping old data from queue (%d bytes)", buf->len);
		// 버퍼는 정적이므로 free하지 않음
	}
}

static void disconnected(struct bt_conn *conn, uint8_t reason)
{
	char addr[BT_ADDR_LE_STR_LEN];

	bt_addr_le_to_str(bt_conn_get_dst(conn), addr, sizeof(addr));

	LOG_INF("Disconnected: %s, reason 0x%02x %s", addr, reason, bt_hci_err_to_str(reason));

	if (auth_conn) {
		bt_conn_unref(auth_conn);
		auth_conn = NULL;
	}

	if (current_conn) {
		bt_conn_unref(current_conn);
		current_conn = NULL;
		connection_ready = false;
        
        // 예약된 작업이 있다면 취소
		k_work_cancel_delayable(&connection_setup_work); 
		
		dk_set_led_off(CON_STATUS_LED);
	}
}

static void bt_receive_cb(struct bt_conn *conn, const uint8_t *const data,
			  uint16_t len)
{
	int err;
	char addr[BT_ADDR_LE_STR_LEN] = {0};

	bt_addr_le_to_str(bt_conn_get_dst(conn), addr, ARRAY_SIZE(addr));

	LOG_INF("Received data from: %s", addr);

	for (uint16_t pos = 0; pos != len;) {
		struct uart_data_t *tx = k_malloc(sizeof(*tx));

		if (!tx) {
			LOG_WRN("Not able to allocate UART send data buffer");
			return;
		}

		/* Keep the last byte of TX buffer for potential LF char. */
		size_t tx_data_size = sizeof(tx->data) - 1;

		if ((len - pos) > tx_data_size) {
			tx->len = tx_data_size;
		} else {
			tx->len = (len - pos);
		}

		memcpy(tx->data, &data[pos], tx->len);

		pos += tx->len;

		/* Append the LF character when the CR character triggered
		 * transmission from the peer.
		 */
		if ((pos == len) && (data[len - 1] == '\r')) {
			tx->data[tx->len] = '\n';
			tx->len++;
		}

		err = uart_tx(uart, tx->data, tx->len, SYS_FOREVER_MS);
		if (err) {
			k_fifo_put(&fifo_uart_tx_data, tx);
		}
	}
}

static void on_nus_send_enabled(enum bt_nus_send_status status)
{
    if (status == BT_NUS_SEND_STATUS_ENABLED) {
        LOG_INF("NUS send enabled: App has SUBSCRIBED. Waiting 100ms for stack to settle...");
        // 플래그를 직접 설정하는 대신 500ms 지연 스케줄링
        k_work_schedule(&connection_setup_work, K_MSEC(500));
    }
}

void error(void)
{
	dk_set_leds_state(DK_ALL_LEDS_MSK, DK_NO_LEDS_MSK);

	while (true) {
		/* Spin for ever */
		k_sleep(K_MSEC(1000));
	}
}

{

	} else {

		LOG_INF("Numeric Reject, conn %p", (void *)auth_conn);

void adc_thread(void)
{
	int err;

	// BLE가 초기화될 때까지 대기
	k_sem_take(&adc_init_ok, K_FOREVER);

	// ADC 장치 준비 확인
	if (!device_is_ready(adc_dev)) {
		LOG_ERR("ADC device not ready");
		return;
	}

	// ADC 채널 설정
	err = adc_channel_setup(adc_dev, &m_channel_cfg);
	if (err) {
		LOG_ERR("Failed to setup ADC channel (err %d)", err);
		return;
	}

	int16_t single_sample_buffer; // 단일 샘플 임시 저장
    const struct adc_sequence single_sample_sequence = {
        .channels = BIT(ADC_CHANNEL_ID),
        .buffer = &single_sample_buffer,
        .buffer_size = sizeof(single_sample_buffer),
        .resolution = ADC_RESOLUTION,
    };

	LOG_INF("ADC Thread started. Waiting for connection...");

	for (;;) {
        // 1. BLE 연결이 완전히 준비되었는지 확인
        if (!connection_ready || !current_conn) {
            
            // 2. 연결이 끊겼으므로, 큐에 쌓인 모든 '오래된' 데이터를 비웁니다.
            //    (다음 연결 시 오래된 데이터가 전송되는 것을 방지)
            struct uart_data_t *buf;
            while ((buf = k_fifo_get(&fifo_uart_rx_data, K_NO_WAIT)) != NULL) {
                // 정적 버퍼 풀은 free하지 않음 (순환 사용)
                LOG_DBG("ADC: Dropping stale packet (disconnected)");
            }
            
            // 3. 연결이 준비될 때까지 200ms간 대기하며 다시 체크
            k_sleep(K_MSEC(200)); 
            continue; // 루프의 처음으로 돌아가 연결 상태 다시 확인
        }

        // 4. (연결됨) 정적 버퍼 풀에서 다음 버퍼 가져오기
		static struct uart_data_t static_buf_stack[10]; // 10개 버퍼 풀
		static int buf_index = 0;
		struct uart_data_t *buf = &static_buf_stack[buf_index];
		buf_index = (buf_index + 1) % 10;

        // 버퍼를 int16_t 포인터로 캐스팅
        int16_t* packet_data = (int16_t*)buf->data;

		// 5. 100Hz 샘플링 (10ms 간격)으로 20개 샘플 수집
        for (int i = 0; i < SAMPLES_PER_PACKET; i++) {
            
            err = adc_read(adc_dev, &single_sample_sequence);
            if (err) {
                LOG_WRN("ADC read sample %d failed (err %d)", i, err);
                packet_data[i] = 0;
            } else {
                packet_data[i] = single_sample_buffer;
            }

            // 100Hz 샘플링 속도를 위해 10ms 대기
            k_sleep(K_MSEC(10));
        }

		// 6. 버퍼 길이 설정 (20 샘플 * 2바이트 = 40 바이트)
		buf->len = SAMPLES_PER_PACKET * sizeof(int16_t);

		// 7. FIFO에 데이터 넣기
		LOG_INF("ADC: Putting %d bytes into FIFO, samples: %d, %d", 
			buf->len, packet_data[0], packet_data[19]);
		k_fifo_put(&fifo_uart_rx_data, buf);
	}
}

	LOG_INF("Bluetooth initialized");

		settings_load();
	}

	LOG_INF("Initializing NUS");
	err = bt_nus_init(&nus_cb);
	if (err) {
		LOG_ERR("Failed to initialize UART service (err: %d)", err);
		return 0;
	}

	LOG_INF("Starting advertising");
	err = bt_le_adv_start(BT_LE_ADV_CONN, ad, ARRAY_SIZE(ad), sd,
			      ARRAY_SIZE(sd));
	if (err) {
		LOG_ERR("Advertising failed to start (err %d)", err);
		return 0;
	}

	LOG_INF("BLE advertising started successfully");

	k_sem_give(&ble_init_ok);

	k_sem_give(&adc_init_ok);
	LOG_INF("ADC Thread semaphore given");

		k_sleep(K_MSEC(RUN_LED_BLINK_INTERVAL));
	}
}

void ble_write_thread(void)
{
    /* Don't go any further until BLE is initialized */
    k_sem_take(&ble_init_ok, K_FOREVER);
    LOG_INF("BLE Write Thread started");

	for (;;) {
		/* Wait indefinitely for data to be sent over bluetooth */
		struct uart_data_t *buf = k_fifo_get(&fifo_uart_rx_data,
						     K_FOREVER);

		LOG_INF("BLE Write: Received %d bytes", buf->len);

		// ADC 데이터는 연결되고 MTU 교환이 완료된 경우에만 전송
		if (current_conn) {
			if (connection_ready) {
				int ret = bt_nus_send(current_conn, buf->data, buf->len);
				if (ret == 0) {
					LOG_DBG("BLE Write: Successfully sent %d bytes", buf->len);
				} else {
					// 실패 시 짧은 재시도 (최대 3번)
					bool retry_success = false;
					for (int i = 0; i < 3 && !retry_success; i++) {
						k_sleep(K_MSEC(10));
						ret = bt_nus_send(current_conn, buf->data, buf->len);
						if (ret == 0) {
							LOG_DBG("BLE Write: Retry successful");
							retry_success = true;
						}
					}
					if (!retry_success) {
						LOG_WRN("BLE Write: Failed after retries (err: %d)", ret);
					}
				}
			} else {
				LOG_INF("BLE Write: Connection not ready (MTU exchange in progress), skipping %d bytes", buf->len);
			}
		} else {
			LOG_DBG("BLE Write: No connection, skipping");
		}

		// 정적 버퍼 풀은 free하지 않음 (순환 사용)
	}
}

		NULL, PRIORITY, 0, 0);
